;;==============================================================================
;; 04_비전_검사_분류
;; 비전 검사 결과 수신 -> OK/NG 래치 -> 리젝트 동기화 -> 불량 배출
;;
;; 처리 내용 :
;;   1. 비전 완료(X5) 상승 엣지 검출 (M60)
;;   2. 판정 코드(D100) 분석 : 0=OK, 1~4=결함 유형, 5~9=기타
;;   3. OK -> D1 증가 / NG -> D2 증가 + 결함 유형별 카운트(D101~D105) + 래치(M20)
;;   4. 리젝트 지연 동기화 : 비전 판정 -> 리젝터 위치 도달까지 D14(10ms) 지연
;;   5. 리젝트 발사 : Y1(에어 블로우) ON, 작동 시간 D15[ms]
;;   6. 미스 리젝트 감지 M23 (분류 구간 재검증 - 센서 추가 시 활성)
;;
;; 보조 디바이스 :
;;   M60 : 비전 완료(X5) 상승 펄스
;;   M61 : D100==0 (OK) CMP 결과
;;   M62 : M51... (사용 안 함 - 예약)
;;   M65 : X5 이전 상태
;;   M70~M72 : CMP(D100, K0) 결과 (M71 = equal)
;;   M73~M75 : CMP(D100, K1) 결과 (M74 = equal) - 유형1
;;   M76~M78 : CMP(D100, K2) 결과 (M77 = equal) - 유형2
;;   M79~M81 : CMP(D100, K3) 결과 (M80 = equal) - 유형3
;;   M82~M84 : CMP(D100, K4) 결과 (M83 = equal) - 유형4
;;   M20 : 불량 래치 (리젝트 대기 중)
;;   M22 : 리젝트 발사 스트로브
;;   M23 : 미스 리젝트 플래그
;;==============================================================================

;;------------------------------------------------------------------------------
;; 네트워크 1 : 비전 완료(X5) 상승 엣지 -> M60 펄스
;;------------------------------------------------------------------------------
        LD      X5
        ANI     M65             ; 이전 OFF 였으면
        OUT     M60             ; 비전 완료 펄스
        LD      X5
        OUT     M65             ; 이전 상태 저장

;;------------------------------------------------------------------------------
;; 네트워크 2 : 판정 코드 비교 #1 - D100 == 0 (OK)?
;;   CMP 결과: M71 = (D100 == K0) 가 ON
;;------------------------------------------------------------------------------
        LD      M60
        CMP     D100    K0      M70     ; M70: D<K0, M71: D=K0, M72: D>K0

;;------------------------------------------------------------------------------
;; 네트워크 3 : OK 처리 - 양품 카운트 D1 증가
;;------------------------------------------------------------------------------
        LD      M60
        AND     M71
        INC     D1              ; 양품 +1

;;------------------------------------------------------------------------------
;; 네트워크 4 : NG 처리 - 불량 래치 M20 세트 + 불량 카운트 D2 증가
;;------------------------------------------------------------------------------
        LD      M60
        ANI     M71             ; OK가 아니면
        SET     M20             ; 불량 래치
        INC     D2              ; 불량 +1

;;------------------------------------------------------------------------------
;; 네트워크 5 : 결함 유형별 카운트 - 유형1 (D100 == 1, 상처/스크래치)
;;------------------------------------------------------------------------------
        LD      M60
        CMP     D100    K1      M73     ; M74 = equal
        LD      M74
        INC     D101            ; 유형1 +1

;;------------------------------------------------------------------------------
;; 네트워크 6 : 결함 유형별 카운트 - 유형2 (D100 == 2, 이물/카본)
;;------------------------------------------------------------------------------
        LD      M60
        CMP     D100    K2      M76     ; M77 = equal
        LD      M77
        INC     D102            ; 유형2 +1

;;------------------------------------------------------------------------------
;; 네트워크 7 : 결함 유형별 카운트 - 유형3 (D100 == 3, 버어/플래시)
;;------------------------------------------------------------------------------
        LD      M60
        CMP     D100    K3      M79     ; M80 = equal
        LD      M80
        INC     D103            ; 유형3 +1

;;------------------------------------------------------------------------------
;; 네트워크 8 : 결함 유형별 카운트 - 유형4 (D100 == 4, 변형/위프)
;;------------------------------------------------------------------------------
        LD      M60
        CMP     D100    K4      M82     ; M83 = equal
        LD      M83
        INC     D104            ; 유형4 +1

;;------------------------------------------------------------------------------
;; 네트워크 9 : 기타 결함 (5~9) -> D105 +1
;;   1~4 어디에도 매치되지 않으면 기타로 분류
;;------------------------------------------------------------------------------
        LD      M60
        ANI     M74             ; 유형1 아님
        ANI     M77             ; 유형2 아님
        ANI     M80             ; 유형3 아님
        ANI     M83             ; 유형4 아님
        INC     D105            ; 기타 +1

;;------------------------------------------------------------------------------
;; 네트워크 10 : 리젝트 지연 타이머 (T7)
;;   불량 래치(M20) 시점 -> 리젝터 위치 도달까지 지연, 설정값 D14[10ms]
;;------------------------------------------------------------------------------
        LD      M20             ; 불량 래치
        ANI     T7
        OUT      T7      D14     ; 지연 시간 (D14 x 10ms)

;;------------------------------------------------------------------------------
;; 네트워크 11 : 리젝트 발사 개시
;;   T7 완료 -> M22 세트 (에어 블로우), 불량 래치 해제
;;------------------------------------------------------------------------------
        LD      T7
        ANI     M3              ; 시퀀스 홀드 시 발사 보류
        SET     M22             ; 리젝트 발사
        RST     M20             ; 불량 래치 해제

;;------------------------------------------------------------------------------
;; 네트워크 12 : 리젝트 발사 출력 (Y1) + 작동 시간 (T3, D15[ms])
;;------------------------------------------------------------------------------
        LD      M22
        OUT      Y1              ; 에어 블로우 ON
        OUT      T3      D15     ; 작동 시간 (D15[ms])
        LD      T3
        RST     M22             ; 발사 종료

;;------------------------------------------------------------------------------
;; 네트워크 13 : 미스 리젝트 감지 (M23) - 예비 로직
;;   분류 구간 도달 센서(X_none) 추가 시: 리젝트 발사 후 제품 미도달 -> 알람
;;   (분류 구간 센서 미설치 시 비활성)
;;------------------------------------------------------------------------------
        LDI     M22             ; 발사 종료 후
        ANI     M23
        OUT     M23             ; 예약 - 필요 시 분류 센서와 결합
