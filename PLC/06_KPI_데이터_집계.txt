;; 06_KPI_데이터_집계
;; 생산 KPI 집계 및 데이터 무결성 검증
;;
;; 처리 내용 :
;;   1. 카운트 무결성 검증 : D0(총생산) == D1(양품) + D2(불량)
;;   2. 결함 유형 합계 산출 : D101+D102+D103+D104+D105 -> D252
;;   3. KPI(불량률/OEE)는 Unity 디지털 트윈에서 계산/표시
;;
;; 보조 디바이스 :
;;   M46  : 카운트 불일치 진단 플래그
;;   M90~M92 : CMP(K0, D251) 결과 (M91 = equal)
;;   D250~D252 : 계산용 스크래치 레지스터
;;==============================================================================

;;------------------------------------------------------------------------------
;; 네트워크 1 : 카운트 무결성 검증
;;   D250 = D0 - D1
;;   D251 = D250 - D2   (0이면 D0 == D1 + D2 정합)
;;------------------------------------------------------------------------------
        LD      M0              ; 운전 중일 때 계산
        SUB     D0      D1      D250    ; D250 = D0 - D1
        SUB     D250    D2      D251    ; D251 = D250 - D2
        CMP     K0      D251    M90     ; M91 = (D251 == 0)
        LDI     M91
        OUT     M46             ; 카운트 불일치 진단 플래그

;;------------------------------------------------------------------------------
;; 네트워크 2 : 결함 유형 합계 산출 (D252)
;;   D252 = D101 + D102 + D103 + D104 + D105
;;   (D252 는 D2 불량 카운트와 비교해 유형 집계 무결성 확인 가능)
;;------------------------------------------------------------------------------
        LD      M0
        ADD     D101    D102    D252    ; 유형1 + 유형2
        ADD     D252    D103    D252    ; + 유형3
        ADD     D252    D104    D252    ; + 유형4
        ADD     D252    D105    D252    ; + 기타

;;------------------------------------------------------------------------------
;; 네트워크 3 : KPI 참조 데이터 (Unity 트윈에서 읽는 대상)
;;   D0  : 총 생산
;;   D1  : 양품
;;   D2  : 불량
;;   D10 : 사이클 타임 [0.1초]
;;   D101~D105 : 결함 유형별 카운트
;;   불량률(%)   = D2 / D0 x 100          -> 트윈 계산
;;   품질률(%)   = D1 / D0 x 100          -> 트윈 계산
;;   OEE         = 가동률 x 성능률 x 품질률 -> 트윈 계산
;;==============================================================================
