;;==============================================================================
;; 05_컨베이어_제어
;; 컨베이어 구동 및 속도 설정
;;
;; 처리 내용 :
;;   1. 컨베이어 모터(Y0) : 운전 중(M0) & 알람 아님 & 홀드 아님 -> 구동
;;   2. 속도 설정(D12) 아날로그 반영 (VFD 연동 - 시스템 구성에 따라 조정)
;;   3. 수동 모드 시 속도만 유지 (자동 시퀀스와 독립)
;;==============================================================================

;;------------------------------------------------------------------------------
;; 네트워크 1 : 컨베이어 모터 구동 (Y0)
;;   운전 중(M0) && 알람 없음(M2 OFF) && 시퀀스 홀드 아님(M3 OFF)
;;------------------------------------------------------------------------------
        LD      M0              ; 운전 중
        ANI     M2              ; 알람 아님
        ANI     M3              ; 홀드 아님
        OUT      Y0              ; 컨베이어 모터

;;------------------------------------------------------------------------------
;; 네트워크 2 : 컨베이어 속도 세팅 반영 (아날로그 출력 예시)
;;   D12[%] -> DA 모듈 변환 레지스터(D_AOUT)로 전송
;;   (실제 DA 모듈 주소는 PLC 구성에 맞게 조정)
;;------------------------------------------------------------------------------
        LD      M0
        MOV     D12     D_AOUT  ; D_AOUT: 실제 DA 모듈 채널 레지스터로 대체

;;------------------------------------------------------------------------------
;; 네트워크 3 : 수동 모드 속도 유지 명령
;;   수동(XA ON) 시에는 조작 패널 세팅값(D12)만 반영, 시퀀스 연동 없음
;;------------------------------------------------------------------------------
        LD      XA              ; 수동 모드
        MOV     D12     D_AOUT  ; 수동 속도 반영 (동일 세팅 사용)

;;==============================================================================
;; END
