;;==============================================================================
;; 03_카운팅_로직
;; 카운팅 센서(X2) 기반 생산 개수 집계 - 채터 필터 + 이중 카운트 방지 + 스틱온 진단
;;
;; 처리 내용 :
;;   1. ON 디바운스 (T0 = 20ms) : 채터링 제거
;;   2. OFF 디바운스 (T1 = 20ms) : 제품 통과 완료 확인
;;   3. 카운트 펄스(M51) 생성 -> D0(총 생산) 1 증가
;;   4. 이중 카운팅 방지 : OFF 확인(M108) 후에만 다음 카운트 허용
;;   5. 스틱온 감지 (T6 = 500ms) : 센서 계속 ON -> M44 알람 + 횟수 기록(D110)
;;
;; 보조 디바이스 :
;;   M50  : 디바운스된 카운팅 센서 신호
;;   M51  : 카운트 1샷 펄스
;;   M52  : 스틱온(T6) 상승 펄스 (D110 기록용)
;;   M107 : M50 이전 상태
;;   M108 : OFF 확인 완료 플래그 (다음 카운트 허용)
;;   M109 : T6 이전 상태
;;==============================================================================

;;------------------------------------------------------------------------------
;; 네트워크 1 : ON 디바운스 (20ms) -> M50
;;   X2가 20ms 이상 ON이면 M50 ON (채터 제거)
;;------------------------------------------------------------------------------
        LD      X2              ; 카운팅 센서
        OUT      T0      K2      ; 20ms
        LD      T0
        OUT     M50             ; 디바운스된 센서 신호

;;------------------------------------------------------------------------------
;; 네트워크 2 : OFF 디바운스 (20ms) -> M108 세트
;;   센서가 20ms 이상 OFF 이면 "제품 통과 완료" 확인 -> 다음 카운트 허용
;;------------------------------------------------------------------------------
        LD      M50
        ANI     X2              ; 센서 OFF
        OUT      T1      K2      ; 20ms
        LD      T1
        SET     M108            ; OFF 확인 완료

;;------------------------------------------------------------------------------
;; 네트워크 3 : 카운트 펄스 생성 (M51)
;;   OFF 확인(M108) 후 첫 ON 감지에서만 1 스캔 펄스
;;------------------------------------------------------------------------------
        LD      M50
        AND     M108
        OUT     M51             ; 카운트 1샷 펄스
        RST     M108            ; 허가 플래그 해제 (OFF 될 때까지 추가 카운트 불가)

;;------------------------------------------------------------------------------
;; 네트워크 4 : 총 생산 카운트 증가 (D0)
;;------------------------------------------------------------------------------
        LD      M51
        INC     D0              ; 총 생산 +1

;;------------------------------------------------------------------------------
;; 네트워크 5 : 스틱온 감지 (500ms 이상 ON 지속) -> M44 알람
;;   T6 = 500ms (M50 ON 지속 시간 측정)
;;------------------------------------------------------------------------------
        LD      M50
        OUT      T6      K50     ; 500ms
        LD      T6
        OR      M44             ; 알람 래치
        ANI     XD              ; 리셋 버튼
        OUT     M44             ; 카운팅 센서 이상 알람

;;------------------------------------------------------------------------------
;; 네트워크 6 : 스틱온 상승 엣지 -> M52 (1회 기록용)
;;------------------------------------------------------------------------------
        LD      T6
        ANI     M109
        OUT     M52             ; T6 상승 펄스
        LD      T6
        OUT     M109            ; 이전 상태 저장

;;------------------------------------------------------------------------------
;; 네트워크 7 : 스틱온 발생 횟수 기록 (D110)
;;------------------------------------------------------------------------------
        LD      M52
        INC     D110            ; 센서 스틱온 횟수 +1

;;==============================================================================
;; END
