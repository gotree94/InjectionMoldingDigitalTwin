;; 01_안전_인터록_메인
;; 사출 공장 디지털 트윈 - 안전 인터록 & 메인 운전 제어
;;
;; 대상 PLC  : 미쓰비시 Q / iQ-R / iQ-F (GX Works2/3)
;; 처리 내용 :
;;   1. 안전 조건(비상정지, 가드 도어, OCR) 검사
;;   2. 자동/수동 절환 + 스타트/스톱 운전 플래그
;;   3. 알람 램프/부저 출력
;;   4. 시퀀스 홀드 처리
;;==============================================================================

;;------------------------------------------------------------------------------
;; 네트워크 1 : 안전 조건 검사 -> 자동 모드 허가 (M1)
;;   [X7 비상정지 NC] [X8 가드도어 NC] [X9 OCR NC]
;;        ---|/|---        ---|/|---        ---|/|---       (M1)
;;                       |----------------------------( )--- 자동모드허가
;;------------------------------------------------------------------------------
        LD      X7              ; 비상정지 (NC -> 정상 시 ON)
        ANI     X8              ; 가드 도어가 닫혀 있어야 함
        ANI     X9              ; 컨베이어 모터 과부하 없음
        AND     XA              ; 절환 스위치: OFF(자동) 상태에서만 허가
        OUT     M1              ; 자동 모드 허가 플래그

;;------------------------------------------------------------------------------
;; 네트워크 2 : 스타트 버튼 -> 운전 플래그 M0 세트
;;   스타트(XB) 순간 펄스 + 자동 모드 허가(M1) + 알람 없음(M2 OFF) -> SET M0
;;------------------------------------------------------------------------------
        LD      XB              ; 스타트 버튼
        AND     M1              ; 자동 모드 허가
        ANI     M2              ; 알람 상태가 아니어야 함
        SET     M0              ; 운전 플래그 세트

;;------------------------------------------------------------------------------
;; 네트워크 3 : 스톱 버튼 -> 운전 플래그 M0 리셋
;;------------------------------------------------------------------------------
        LD      XC              ; 스톱 버튼
        OR      M2              ; 또는 알람 발생 시
        RST     M0              ; 운전 플래그 리셋

;;------------------------------------------------------------------------------
;; 네트워크 4 : 알람 조건 ⓐ 비상정지
;;   X7(NC) OFF 시 알람 발생 -> M40 세트, 운전 정지
;;------------------------------------------------------------------------------
        LDI     X7              ; 비상정지 버튼 눌림 (NC 해제)
        OR      M40             ; 알람 유지(래치)
        ANI     XD              ; 리셋 버튼으로 해제 가능
        OUT     M40             ; 비상정지 알람

;;------------------------------------------------------------------------------
;; 네트워크 5 : 알람 조건 ⓑ 가드 도어
;;------------------------------------------------------------------------------
        LDI     X8              ; 가드 도어 열림
        OR      M41
        ANI     XD
        OUT     M41             ; 가드 도어 알람

;;------------------------------------------------------------------------------
;; 네트워크 6 : 알람 조건 ⓒ 컨베이어 과부하
;;------------------------------------------------------------------------------
        LDI     X9              ; OCR 동작
        OR      M42
        ANI     XD
        OUT     M42             ; 과부하 알람

;;------------------------------------------------------------------------------
;; 네트워크 7 : 알람 집합 (M40~M45 -> M2 알람 플래그)
;;------------------------------------------------------------------------------
        LD      M40
        OR      M41
        OR      M42
        OR      M43             ; 불량 박스 풀 알람(이후 블록에서 세트)
        OR      M44             ; 카운팅 센서 이상 알람(03번 파일에서 세트)
        OR      M45             ; 이젝션 타임아웃 알람(02번 파일에서 세트)
        OUT     M2              ; 알람 플래그

;;------------------------------------------------------------------------------
;; 네트워크 8 : 알람 코드 작성 (D900)
;;   0=정상, 1=비상정지, 2=가드도어, 3=과부하, 4=불량박스풀, 5=센서이상, 6=타임아웃
;;------------------------------------------------------------------------------
        LD      M41             ; 가드 도어
        MOV     K2      D900    ; D900 = 2
        LD      M42             ; 과부하
        MOV     K3      D900    ; D900 = 3
        LD      M43
        MOV     K4      D900
        LD      M44
        MOV     K5      D900
        LD      M45
        MOV     K6      D900
        LD      M40
        MOV     K1      D900
        LDI     M2
        MOV     K0      D900

;;------------------------------------------------------------------------------
;; 네트워크 9 : 시퀀스 홀드 M3
;;   알람(M2) 또는 불량박스 풀(X6) 시 시퀀스 멈춤
;;------------------------------------------------------------------------------
        LD      M2              ; 알람 플래그
        OR      X6              ; 불량 박스 풀 센서
        OR      M43             ; 불량박스풀 알람 래치
        OUT     M3              ; 시퀀스 홀드

;;------------------------------------------------------------------------------
;; 네트워크 10 : 런닝 램프 (자동 운전 중)
;;------------------------------------------------------------------------------
        LD      M0
        ANI     M2
        OUT      Y3              ; 운전 램프

;;------------------------------------------------------------------------------
;; 네트워크 11 : 알람 램프
;;------------------------------------------------------------------------------
        LD      M2
        OUT      Y4              ; 알람 램프

;;------------------------------------------------------------------------------
;; 네트워크 12 : 부저 (0.5초 ON / 0.5초 OFF 점멸)
;;   T10 = ON 주기 타이머 / T11 = OFF 주기 타이머
;;------------------------------------------------------------------------------
        LD      M2
        AND     T11
        OUT      T10     K5      ; 0.5초 ON
        LD      M2
        ANI     T10
        OUT      T11     K5      ; 0.5초 OFF
        LD      T10
        OUT      Y5              ; 부저
