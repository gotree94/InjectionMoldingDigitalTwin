;;==============================================================================
;; 02_사출_이젝션_시퀀스
;; 사출 완료 -> 이젝션 -> 컨베이어 적재(카운팅 대기) 시퀀스
;;
;; 처리 내용 :
;;   1. 사출 완료(X0) 상승 엣지 검출
;;   2. 스텝 시프트: IDLE -> EJECT -> COUNT_WAIT -> IDLE
;;   3. 이젝션 개시(Y6), 이젝션 완료(X1) 감시 + 타임아웃
;;   4. 사이클 타임 측정 (D10 [0.1초])
;;   5. 샷 카운터 증가 (D33, 금형 수명 관리)
;;
;; 보조 디바이스 :
;;   M100 : X0 상승 펄스
;;   M101 : X0 이전 상태
;;   M105 : T12(0.1초 타이머) 상승 펄스
;;   M106 : T12 이전 상태 저장
;;   D200 : 사이클 측정 누적 [0.1초 단위]
;;==============================================================================

;;------------------------------------------------------------------------------
;; 네트워크 1 : 사출 완료(X0) 상승 엣지 검출 -> M100 펄스
;;------------------------------------------------------------------------------
        LD      X0              ; 사출 완료
        ANI     M101            ; 직전 OFF 였으면
        OUT     M100            ; 상승 펄스
        LD      X0
        OUT     M101            ; 이전 상태 저장

;;------------------------------------------------------------------------------
;; 네트워크 2 : 스텝 0 (M10 STEP_IDLE) 조건 -> 스텝 1 (M11 STEP_EJECT)
;;   사출 완료 + 운전 중 + 홀드 아님 -> 이젝션 스텝
;;------------------------------------------------------------------------------
        LD      M10             ; 대기 스텝
        AND     M0              ; 운전 중
        AND     M100            ; 사출 완료 상승
        ANI     M3              ; 시퀀스 홀드 아님
        SET     M11             ; 이젝션 스텝 진입
        RST     M10             ; 대기 스텝 해제

;;------------------------------------------------------------------------------
;; 네트워크 3 : 스텝 1 (M11) -> 이젝션 개시 지령 (Y6)
;;------------------------------------------------------------------------------
        LD      M11
        OUT      Y6              ; 이젝터 상승 지령

;;------------------------------------------------------------------------------
;; 네트워크 4 : 이젝션 완료 감시 타이머 (T2 = 3초 타임아웃)
;;   완료(X1) 안 오면 타임아웃 -> 알람 M45
;;------------------------------------------------------------------------------
        LD      M11
        ANI     X1              ; 이젝션 완료 아직 안 옴
        OUT      T2      K30     ; 3초 대기
        LD      T2
        OR      M45             ; 알람 래치
        ANI     XD              ; 리셋 버튼
        OUT     M45             ; 타임아웃 알람

;;------------------------------------------------------------------------------
;; 네트워크 5 : 이젝션 완료(X1) 수신 -> 스텝 2 (M12 STEP_COUNT_WAIT)
;;------------------------------------------------------------------------------
        LD      M11
        AND     X1              ; 이젝션 완료
        ANI     M3              ; 홀드 아님
        SET     M12             ; 카운팅 대기 진입
        RST     M11             ; 이젝션 스텝 해제

;;------------------------------------------------------------------------------
;; 네트워크 6 : 스텝 2 (M12) - 컨베이어 적재 대기
;;   카운트 펄스(M51, 03번 파일) 수신 시 대기 스텝(M10)으로 복귀
;;------------------------------------------------------------------------------
        LD      M12
        AND     M51             ; 카운팅 펄스 (03번 파일에서 생성)
        ANI     M3
        SET     M10             ; 대기 스텝 복귀
        RST     M12

;;------------------------------------------------------------------------------
;; 네트워크 7 : 사출 완료 시 샷 카운터 증가 (금형 수명 관리)
;;------------------------------------------------------------------------------
        LD      M100
        INC     D33             ; 샷 카운터 +1

;;------------------------------------------------------------------------------
;; 네트워크 8 : 사이클 타임 측정 - 0.1초 자기 발진 타이머 (T12)
;;   LD M0 / ANI T12 / OUT T12 K1 -> 0.1초 주기 ON/OFF 반복
;;------------------------------------------------------------------------------
        LD      M0              ; 운전 중
        ANI     T12
        OUT      T12     K1      ; 0.1초

;;------------------------------------------------------------------------------
;; 네트워크 8b : T12 상승 엣지 검출 (M105)
;;------------------------------------------------------------------------------
        LD      T12
        ANI     M106            ; 이전 상태 OFF 였으면
        OUT     M105            ; 상승 펄스
        LD      T12
        OUT     M106            ; 이전 상태 저장

;;------------------------------------------------------------------------------
;; 네트워크 8c : 0.1초마다 D200 증가 (사이클 누적)
;;------------------------------------------------------------------------------
        LD      M105
        INC     D200

;;------------------------------------------------------------------------------
;; 네트워크 9 : 사출 완료(X0) 시 사이클 타임 저장 및 리셋
;;------------------------------------------------------------------------------
        LD      M100
        MOV     D200    D10     ; 직전 사이클 타임 [0.1초] 저장
        MOV     K0      D200    ; 누적 리셋
